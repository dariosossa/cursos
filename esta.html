Conceptos Fundamentales

Estadística Descriptiva: Rama encargada de recopilar, organizar, resumir y presentar conjuntos de datos numéricos en tablas o gráficos precisos.

Población: Conjunto total de individuos, objetos o medidas que poseen características comunes observables.

Muestra: Subconjunto representativo extraído de una población para realizar estudios analíticos y poblacionales.

Frecuencia Absoluta ($f_i$): Número de veces que se repite un determinado valor o intervalo en el conjunto de datos.

Marca de Clase ($x_i$): Valor representativo de un intervalo, calculado como el promedio entre su límite inferior y superior.

EstadisticaBasica.java 
/* ==========================================
 * EJERCICIO 1: Media y Varianza Poblacional
 * Datos de entrada: 10.0, 15.0, 20.0
 * ========================================== */

public class EstadisticaBasica {
    
    public static void main(String[] args) {
        double[] notas = {10.0, 15.0, 20.0};
        
        if (notas == null || notas.length == 0) {
            System.out.println("Error: Sin datos.");
            return;
        }
        
        double media = calcularMedia(notas);
        double varianza = calcularVarianza(notas, media);
        
        System.out.println("Media: " + media);
        System.out.println("Varianza: " + varianza);
    }

    public static double calcularMedia(double[] d) {
        double s = 0.0;
        for (double val : d) s += val;
        return s / d.length;
    }

    public static double calcularVarianza(double[] d, double m) {
        double sd = 0.0;
        for (double val : d) sd += Math.pow(val - m, 2);
        return sd / d.length;
    }
}
CalculadorMediana.java 
/* ==========================================
 * EJERCICIO 2: Algoritmo para Mediana
 * Ordenamiento previo requerido
 * ========================================== */
import java.util.Arrays;

public class CalculadorMediana {
    public static void main(String[] args) {
        double[] datos = {7.0, 3.0, 9.0, 1.0, 5.0};
        Arrays.sort(datos);
        
        double mediana;
        int n = datos.length;
        if (n % 2 == 0) {
            mediana = (datos[n/2 - 1] + datos[n/2]) / 2.0;
        } else {
            mediana = datos[n / 2];
        }
        System.out.println("Mediana calculada: " + mediana);
    }
}

/* ==========================================
 * EJERCICIO 3: Cálculo del Punto de Equilibrio
 * Costos Fijos / (Precio - Costo Variable)
 * ========================================== */
public class PuntoEquilibrio {
    public static void main(String[] args) {
        double costosFijos = 5000.0;
        double precioVentaUnitario = 50.0;
        double costoVariableUnitario = 30.0;
        
        double unidadesEq = costosFijos / (precioVentaUnitario - costoVariableUnitario);
        System.out.println("Unidades en Equilibrio: " + unidadesEq);
    }
}
